# PowerShell 7: Active Directory Hardening & NTFS Inheritance Stripping
function Set-SecureDepartmentShare {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$FolderPath,
        [Parameter(Mandatory)]
        [string]$DepartmentGroup
    )

    Write-Host "[+] Hardening NTFS permissions on $FolderPath" -ForegroundColor Cyan
    
    # Disable inheritance and remove default inherited broad permissions
    $acl = Get-Acl -Path $FolderPath
    $acl.SetAccessRuleProtection($true, $false) # (isProtected, preserveInheritance)

    # Define explicit Least-Privilege access rules
    $adminRule = [System.Security.AccessControl.FileSystemAccessRule]::new(
        "BUILTIN\Administrators", "FullControl", 
        "ContainerInherit,ObjectInherit", "None", "Allow"
    )
    $deptRule = [System.Security.AccessControl.FileSystemAccessRule]::new(
        "CORP\$DepartmentGroup", "Modify,Synchronize",
        "ContainerInherit,ObjectInherit", "None", "Allow"
    )

    $acl.AddAccessRule($adminRule)
    $acl.AddAccessRule($deptRule)
    Set-Acl -Path $FolderPath -AclObject $acl

    Write-Host "[✓] Applied CIS baseline access controls: CORP\$DepartmentGroup" -ForegroundColor Green
}
